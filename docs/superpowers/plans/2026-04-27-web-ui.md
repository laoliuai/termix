# Web UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the browser-first Web UI per the approved design — a three-screen Preact SPA (Login = landing / SessionList / Terminal) with HttpOnly-cookie refresh + in-memory access-token auth, served by `termix-control` via `go:embed`, installable as a PWA on Android Chrome and iOS Safari, deployable to a public domain on a single binary.

**Architecture:** Slice-1 `android/terminal-web/` is promoted in place to `web/app/`. The existing `protocol/net/session/ui` modules carry their 69 unit tests forward unchanged (one row added for `C-j`). On top, a Preact + `@preact/signals` + `preact-router` SPA owns Login + SessionList screens and wraps the existing terminal-web runtime as the Terminal page. Backend gets four contract-first additions in the same slice: dual-mode `/v1/auth/login` and `/v1/auth/refresh` (cookie when `device_type=web`, body otherwise — Compose stays unchanged), new `POST /v1/auth/logout`, plus security-headers / login-rate-limit / Origin-allowlist middlewares, a `go:embed` static handler, an access-token TTL revert, and a relay log query-string scrub.

**Tech Stack:** Preact 10, `@preact/signals` 1.2, `preact-router` 4, Vite 5, `vite-plugin-pwa` (Workbox), Vitest + happy-dom + `@preact/testing-library`, xterm.js (existing); Go 1.25 with Gin (existing), `oapi-codegen`, `sqlc + pgx`, `golang-migrate`, `golang.org/x/time/rate` for rate limiting.

**Authoritative spec:** `docs/superpowers/specs/2026-04-27-web-ui-design.md`.

**Worktree convention:** Per AGENTS.md, work this slice in `.worktrees/web-ui/` on a branch `web-ui`. Create with `git worktree add .worktrees/web-ui -b web-ui`. All work lands on this branch; merge to `main` only after the slice is complete.

func (s *Server) PostAuthRefresh(c *gin.Context) {
    var rawToken string
    if ck, err := c.Request.Cookie("termix_refresh"); err == nil && ck.Value != "" {
        rawToken = ck.Value
    } else {
        var req openapi.RefreshRequest
        _ = c.ShouldBindJSON(&req) // body optional
        if req.RefreshToken != nil {
            rawToken = *req.RefreshToken
        }
    }
    if rawToken == "" {
        c.JSON(http.StatusUnauthorized, gin.H{"error": "missing refresh credential"})
        return
    }

    row, err := s.tokens.LookupRefresh(c.Request.Context(), rawToken)
    if err != nil {
        c.JSON(http.StatusUnauthorized, gin.H{"error": "invalid refresh"})
        return
    }
    accessTok, err := s.tokens.MintAccess(row.UserID, row.DeviceID)
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": "mint access"})
        return
    }
    c.JSON(http.StatusOK, openapi.RefreshResponse{
        AccessToken: accessTok.Value,
        ExpiresInSeconds: int(accessTok.TTL.Seconds()),
        RefreshToken: nil, // V1 no rotation
    })
}

```go
package middleware

import (
    "bytes"
    "encoding/json"
    "io"
    "net"
    "net/http"
    "sync"
    "time"

    "github.com/gin-gonic/gin"
    "golang.org/x/time/rate"
)

type bucket struct {
    lim *rate.Limiter
    last time.Time
}

type loginLimiter struct {
    mu      sync.Mutex
    buckets map[string]*bucket
    refill  rate.Limit
    burst   int
    ttl     time.Duration
}

func LoginRateLimit(perMinute, burst int) gin.HandlerFunc {
    l := &loginLimiter{
        buckets: map[string]*bucket{},
        refill: rate.Every(time.Minute / time.Duration(perMinute)),
        burst: burst,
        ttl: 5 * time.Minute,
    }
    return func(c *gin.Context) {
        // Peek email from body without consuming the stream.
        var email string
        if c.Request.Body != nil {
            buf, _ := io.ReadAll(c.Request.Body)
            c.Request.Body = io.NopCloser(bytes.NewReader(buf))
            var probe struct{ Email string `json:"email"` }
            _ = json.Unmarshal(buf, &probe)
            email = probe.Email
        }

        ip, _, _ := net.SplitHostPort(c.Request.RemoteAddr)
        key := ip + "|" + email

        l.mu.Lock()
        b, ok := l.buckets[key]
        if !ok {
            b = &bucket{lim: rate.NewLimiter(l.refill, l.burst)}
            l.buckets[key] = b
        }
        b.last = time.Now()
        // sweep stale buckets opportunistically
        for k, bb := range l.buckets {
            if time.Since(bb.last) > l.ttl {
                delete(l.buckets, k)
            }
        }
        allowed := b.lim.Allow()
        l.mu.Unlock()

        if !allowed {
            c.AbortWithStatusJSON(http.StatusTooManyRequests, gin.H{"error": "too many login attempts"})
            return
        }
        c.Next()
    }
}
```

func StaticHandler() gin.HandlerFunc {

    if devDir := os.Getenv("TERMIX_CONTROL_WEB_DIR"); devDir != "" {
        fsys = os.DirFS(devDir)

import { accessToken, accessTokenExpiresAt } from "./store";

let inflight: Promise<string | null> | null = null;

export async function doRefreshOnce(): Promise<string | null> {
  if (inflight) return inflight;
  inflight = (async (): Promise<string | null> => {
    // fast path: someone already refreshed
    if (accessToken.value && Date.now() < accessTokenExpiresAt.value - 5_000) {
      return accessToken.value;
    }
    try {
      const res = await fetch("/api/v1/auth/refresh", { method: "POST" });
      if (!res.ok) return null;
      const data = await res.json();
      accessToken.value = data.access_token;
      accessTokenExpiresAt.value = Date.now() + data.expires_in_seconds * 1000;
      return data.access_token;
    } catch {
      return null;
    } finally {
      inflight = null;
    }
  })();
  return inflight;
}

export async function freshAccessToken(): Promise<string | null> {
  if (accessToken.value && Date.now() < accessTokenExpiresAt.value - 60_000) {
    return accessToken.value;
  }
  return doRefreshOnce();
}

// test-only helper
export function __resetInflight(): void { inflight = null; }

import { accessToken } from "../auth/store";
import { doRefreshOnce } from "../auth/refresh";

export async function fetchWithAuth(input: RequestInfo, init: RequestInit = {}): Promise<Response> {
  const doFetch = (token: string | null): Promise<Response> => fetch(input, {
    ...init,
    headers: {
      ...(init.headers ?? {}),
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
  });

  const first = await doFetch(accessToken.value);
  if (first.status !== 401) return first;

  const refreshed = await doRefreshOnce();
  if (refreshed === null) return first;
  return doFetch(refreshed);
}

export function useVisibility(onShow: () => void): void {
  useEffect(() => {
    const handler = () => {
      if (document.visibilityState === "visible") onShow();
    };
    document.addEventListener("visibilitychange", handler);
    return () => document.removeEventListener("visibilitychange", handler);

    const { result } = renderHook(() => useKeyboardOffset());

import { useEffect } from "preact/hooks";
import { signal, type Signal } from "@preact/signals";

export function useKeyboardOffset(): Signal<number> {
  const offset = signal(0);
  useEffect(() => {
    const vv = (window as any).visualViewport as VisualViewport | undefined;
    if (!vv) return;
    const update = () => { offset.value = window.innerHeight - vv.height - vv.offsetTop; };
    vv.addEventListener("resize", update);
    vv.addEventListener("scroll", update);
    update();
    return () => {
      vv.removeEventListener("resize", update);
      vv.removeEventListener("scroll", update);
    };
  }, []);
  return offset;
}

  const email = useSignal("");
  const password = useSignal("");
  const busy = useSignal(false);
  const error = useSignal<string | null>(null);

  const submit = async (e: Event) => {
    e.preventDefault();

    busy.value = true; error.value = null;
    try {
      const res = await login({
        email: email.value, password: password.value,
        device_label: navigator.userAgent.slice(0, 80),
      });
      if (!res.ok) {
        error.value =
          res.status === 401 ? "邮箱或密码错误" :
          res.status === 429 ? "尝试过于频繁，请稍候" :
          res.status === 0 || res.message === "Failed to fetch" ? "无法连接服务器" :
          (res.message || "登录失败");

        return;

      accessTokenExpiresAt.value = Date.now() + res.data.expires_in_seconds * 1000;
      userInfo.value = { user: res.data.user, device: res.data.device };
      onSuccess();

      busy.value = false;
    }
  };

declare global {
  interface Window {
    setSession: (id: string, url: string, tok: string, dev: string) => void;
    sendText: (s: string) => void;
    sendSpecialKey: (k: SpecialKey) => void;
    requestControl: () => void;
    releaseControl: () => void;
    TermixBridge?: {
      onConnectionState: (s: string, d?: string | null) => void;
      onControlState: (s: string, d?: string | null) => void;
      onError: (code: string, msg: string) => void;
    };
  }

export interface TerminalPageProps {
  sessionId: string;
  onBack: () => void;
}

export function TerminalPage({ sessionId, onBack }: TerminalPageProps) {
  const connState = useSignal<"connecting" | "connected" | "disconnected" | "error">("connecting");
  const controlState = useSignal<"none" | "requesting" | "granted" | "denied" | "revoked">("none");
  const keyboardOffset = useKeyboardOffset();

  useEffect(() => {
    let retried = false;
    window.TermixBridge = {
      onConnectionState: (s) => { connState.value = s as any; },
      onControlState: (s) => { controlState.value = s as any; },
      onError: async (code, msg) => {
        if (code === "auth" && !retried) {
          retried = true;
          notify("会话过期，正在刷新…", "warn");
          const tok = await freshAccessToken();
          if (tok && userInfo.value) {
            window.setSession(sessionId, import.meta.env.VITE_RELAY_WS_URL, tok, userInfo.value.device.id);
            return;
          }
          notify("会话已过期，请重新登录", "error");
          onBack();
          return;
        }
        if (code === "watch") notify(msg, "error");
      },
    };

    (async () => {
      const tok = await freshAccessToken();
      if (!tok || !userInfo.value) {
        notify("会话已过期，请重新登录", "warn");
        onBack();
        return;
      }
      window.setSession(sessionId, import.meta.env.VITE_RELAY_WS_URL, tok, userInfo.value.device.id);
    })();

    return () => {
      window.setSession("", "", "", "");
      delete window.TermixBridge;
    };
  }, [sessionId]);

  const onDigit = (d: string) => window.sendText(d);
  const onSpecial = (k: SpecialKey) => window.sendSpecialKey(k);
  const onCompose = (s: string) => window.sendText(s);

  const disabled = controlState.value !== "granted";

  return (
    <div class="terminal-page" style={{ paddingBottom: `${keyboardOffset.value}px` }}>
      <div class="term-header">
        <button class="back" aria-label="back" onClick={onBack}>‹</button>
        <div class="meta">
          <div class="name">session {sessionId.slice(0, 8)}</div>
        </div>
        <span class={`badge conn-${connState.value}`}>{connState.value}</span>
      </div>
      <div class="control-bar">
        <span class={`ctrl-state ctrl-${controlState.value}`}>● {controlLabel(controlState.value)}</span>
        {controlState.value === "granted" ? (
          <button class="release-btn" onClick={() => window.releaseControl()}>Release</button>
        ) : (
          <button class="request-btn" onClick={() => window.requestControl()}>Request Control</button>
        )}
      </div>
      <div id="terminal" class="terminal-host"></div>
      <Composer disabled={disabled} onSend={onCompose} placeholder="Type and Send..." />
      <Toolbar disabled={disabled} onDigit={onDigit} onSpecial={onSpecial} />
    </div>
  );
}

function controlLabel(s: string): string {
  switch (s) {
    case "granted": return "You have control";
    case "requesting": return "Requesting…";
    case "denied": return "Control denied";
    case "revoked": return "Control revoked";
    default: return "Read-only";
  }
}
